### 关键设计思想
**把计算交给代码，把创意交给模型**：
- 所有精确的数值计算（热量、嘌呤、营养成分）都由APP本地代码完成
- 所有需要创意和组合的工作（菜品搭配、口味调整）由本地模型完成
- 所有超出本地模型能力的复杂情况由在线模型兜底

### 3. Qwen2分词器实现 (`Qwen2Tokenizer.kt`)
```kotlin
import android.content.Context
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import java.nio.charset.StandardCharsets

class Qwen2Tokenizer(context: Context) {
    private val vocab: Map<String, Int>
    private val reverseVocab: Map<Int, String>

    init {
        val vocabJson = context.assets.open("qwen2-vocab.json")
            .bufferedReader(StandardCharsets.UTF_8)
            .readText()
        
        val type = object : TypeToken<Map<String, Int>>() {}.type
        vocab = Gson().fromJson(vocabJson, type)
        reverseVocab = vocab.entries.associate { it.value to it.key }
    }

    fun encode(text: String): LongArray {
        val tokens = mutableListOf<Long>()
        tokens.add(151644) // <|im_start|>
        tokens.add(151645) // <|im_end|> 开头标记
        
        // 简单的BPE编码实现（实际项目中建议使用完整的BPE库）
        var i = 0
        while (i < text.length) {
            var j = text.length
            while (j > i) {
                val substr = text.substring(i, j)
                if (vocab.containsKey(substr)) {
                    tokens.add(vocab[substr]!!.toLong())
                    i = j
                    break
                }
                j--
            }
            if (j == i) {
                // 处理未知字符
                tokens.add(vocab["<|unk|>"]!!.toLong())
                i++
            }
        }
        
        tokens.add(151645) // <|im_end|> 结尾标记
        return tokens.toLongArray()
    }

    fun decode(tokens: LongArray): String {
        val sb = StringBuilder()
        for (token in tokens) {
            val tokenId = token.toInt()
            if (tokenId == 151644 || tokenId == 151645) continue
            if (reverseVocab.containsKey(tokenId)) {
                sb.append(reverseVocab[tokenId])
            }
        }
        return sb.toString()
    }
}
```

### 4. 安全校验层 (`utils/RecommendationValidator.kt`)
```kotlin
object RecommendationValidator {
    fun validate(recommendation: DishRecommendation, userProfile: UserProfile): Boolean {
        // 检查是否包含过敏食材
        if (recommendation.ingredients.any { it.name in userProfile.allergies }) {
            return false
        }
        
        // 检查是否包含不喜欢的食材
        if (recommendation.ingredients.any { it.name in userProfile.dislikes }) {
            return false
        }
        
        // 检查嘌呤含量
        if ("gout" in userProfile.diseases && recommendation.total_purine > userProfile.maxPurinePerMeal) {
            return false
        }
        
        // 检查热量
        if (recommendation.total_calories > userProfile.maxCaloriesPerMeal) {
            return false
        }
        
        // 检查烹饪时间
        if (recommendation.cooking_time > 30) {
            return false
        }
        
        return true
    }

    fun extractJsonFromResponse(response: String): String? {
        // 从模型输出中提取JSON部分
        val startIndex = response.indexOf('{')
        val endIndex = response.lastIndexOf('}')
        
        if (startIndex == -1 || endIndex == -1 || endIndex <= startIndex) {
            return null
        }
        
        return response.substring(startIndex, endIndex + 1)
    }
}
```

### 5. 模型推理核心 (`MenuRecommender.kt`)
```kotlin
import android.content.Context
import com.google.gson.Gson
import com.microsoft.onnxruntime.OrtEnvironment
import com.microsoft.onnxruntime.OrtSession
import java.util.*

class MenuRecommender(private val context: Context) {
    private lateinit var ortSession: OrtSession
    private lateinit var tokenizer: Qwen2Tokenizer
    private val gson = Gson()

    // 单例模式
    companion object {
        @Volatile
        private var instance: MenuRecommender? = null

        fun getInstance(context: Context): MenuRecommender {
            return instance ?: synchronized(this) {
                instance ?: MenuRecommender(context.applicationContext).also {
                    instance = it
                    it.initialize()
                }
            }
        }
    }

    private fun initialize() {
        // 初始化分词器
        tokenizer = Qwen2Tokenizer(context)
        
        // 初始化ONNX Runtime
        val env = OrtEnvironment.getEnvironment()
        val sessionOptions = OrtSession.SessionOptions()
        
        // 设置线程数（根据手机CPU核心数调整）
        sessionOptions.setIntraOpNumThreads(Runtime.getRuntime().availableProcessors())
        sessionOptions.setInterOpNumThreads(1)
        
        // 启用CPU优化
        sessionOptions.setGraphOptimizationLevel(OrtSession.SessionOptions.OptLevel.ALL_OPT)
        
        // 加载模型
        val modelBytes = context.assets.open("qwen2.5-0.5b-instruct-q4.onnx").readBytes()
        ortSession = env.createSession(modelBytes, sessionOptions)
    }

    fun recommend(
        userProfile: UserProfile,
        mealType: String,
        history: List<String>,
        maxRetries: Int = 3
    ): List<DishRecommendation> {
        val prompt = PromptTemplates.buildUserPrompt(userProfile, mealType, history)
        val inputIds = tokenizer.encode(prompt)
        
        repeat(maxRetries) { attempt ->
            try {
                // 准备输入
                val inputs = mutableMapOf<String, Any>()
                inputs["input_ids"] = arrayOf(inputIds)
                
                // 运行推理
                val outputs = ortSession.run(inputs)
                val outputIds = outputs["output_ids"] as Array<LongArray>
                
                // 解码输出
                val response = tokenizer.decode(outputIds[0])
                
                // 提取并解析JSON
                val jsonStr = RecommendationValidator.extractJsonFromResponse(response)
                    ?: throw Exception("无法提取JSON")
                
                val result = gson.fromJson(jsonStr, RecommendationResult::class.java)
                
                // 验证推荐结果
                val validRecommendations = result.recommendations.filter {
                    RecommendationValidator.validate(it, userProfile)
                }
                
                if (validRecommendations.isNotEmpty()) {
                    return validRecommendations
                }
                
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
        
        throw Exception("经过$maxRetries次尝试，未能生成有效的推荐结果")
    }

    fun close() {
        ortSession.close()
    }

    private data class RecommendationResult(
        val recommendations: List<DishRecommendation>
    )
}
```

## 七、常见问题解决

1. **模型加载失败**：
   - 确保模型文件在assets目录中
   - 检查文件大小是否正确（约380MB）
   - 确保AndroidManifest.xml中没有添加压缩排除规则

2. **推理速度慢**：
   - 确保使用的是arm64-v8a架构
   - 减少上下文长度（从4096改为2048）
   - 使用更低量化级别的模型（如IQ3_M）

3. **输出格式不稳定**：
   - 优化系统提示词，强调JSON格式要求
   - 在安全校验层添加更严格的JSON提取逻辑
   - 考虑使用引导生成技术（guided generation）
